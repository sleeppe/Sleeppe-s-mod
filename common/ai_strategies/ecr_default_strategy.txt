INJECT:ai_strategy_default = {
	diplomatic_play_support = {
		### The Anglo Fun Police
		if = {
			limit = {
				root = {
					country_rank >= rank_value:great_power
					OR = {
						c:GBR ?= this
						is_in_geographic_region = geographic_region_americas
					}
				}
				scope:enemy_country = {
					has_journal_entry = je_reconquista
					exists = scope:is_initiator
				}
				scope:country = {
					is_historical_spanish_american_colony = yes
				}
			}
			add = {
				value = 50
				desc = "je_reconquista"
			}
		}
	}
	
	obligation_value = {
		# Bonus for obligations from Great Powers
		if = {
			limit = { scope:target_country.country_rank >= rank_value:great_power }
			add = 5
		}
	}

	aggression = {
		# Very high aggression against pariah countries by Great Powers
		if = {
			limit = {
				country_rank >= rank_value:great_power
				scope:target_country = {
					infamy >= infamy_threshold:pariah
					NOT = {
						any_diplomatic_play = {
							is_diplomatic_play_type = dp_contain_threat
							target_is = scope:target_country
						}
					}
				}
			}
			add = 100
		}

		# Russia should very aggressively go for the Caucasus if it's still a great power.
		if = {
			limit = {
				exists = c:RUS
                c:RUS ?= THIS
                country_rank >= rank_value:great_power
                OR = {
                	exists = c:CIR
                	exists = c:CHC
                }
				scope:target_country = {
					NOT = {
						is_country_type = recognized
					}
					OR = {
						has_state_in_state_region = STATE_KUBAN
						has_state_in_state_region = STATE_DAGESTAN
						has_state_in_state_region = STATE_NORTH_CAUCASUS
					}
				}
				NOT = {
					has_global_variable = caucasus_failure_var
				}
			}
			add = 10
		}

		if = { # Spanish American reconquest of former colonies
			limit = {
				has_journal_entry = je_reconquista
				scope:target_country = {
					is_historical_spanish_american_colony = yes
					NOT = { is_subject_of = root }
				}
			}
			add = 2.5

			if = {
				limit = {
					country_rank >= rank_value:great_power
				}
				add = 2.5
			}
		}

	}

	strategic_region_stance_type_scores = {
		stance_conquer_region = {
			if = {
				limit = {
					scope:target_region = {
						any_scope_country = {
							NOR = {
								this = root
								is_subject_of = root
							}
						}
					}
				}

				add = {
					desc = "AI_REGION_STANCE_BASE_VALUE"
					value = 100
				}

				add = {
					desc = "AI_REGION_STANCE_NUMBER_OF_CLAIMS"
					scope:target_region = {
						every_scope_state = {
							limit = {
								owner = {
									NOT = { is_country_type = decentralized }
								}
								has_claim_by = root
							}
							add = 100
						}
					}
				}

				if = {
					limit = {
						is_country_type = unrecognized
						scope:target_region = {
							this = root.capital.region
						}
					}
					add = {
						desc = "AI_REGION_STANCE_CAPITAL_REGION"
						value = 150
					}
				}

				if = {
					limit = {
						scope:target_region = {
							this = sr:region_northeast_asia
						}
					}

					if = {
						limit = {
							country_rank >= rank_value:great_power
							navy_size >= 25
							has_technology_researched = nationalism
							c:JAP ?= {
								has_law_or_variant = law_type:law_isolationism
							}
						}
						add = {
							desc = "AI_REGION_STANCE_WANTS_TO_OPEN_UP_JAPAN"
							value = 1000
						}
					}
				}

				if = {
					limit = {
						scope:target_region = {
							this = sr:region_western_europe
						}
					}

					if = {
						limit = {
							has_strategy = ai_strategy_restore_natural_borders
						}
						add = {
							desc = "ai_strategy_restore_natural_borders"
							add = 1000
						}
					}
				}

				if = {
					limit = {
						has_strategy = ai_strategy_territorial_expansion
					}
					multiply = {
						desc = "ai_strategy_territorial_expansion"
						value = 3
					}
				}

				if = {
					limit = {
						is_subject = yes
						liberty_desire <= 25
					}
					multiply = {
						desc = "AI_REGION_STANCE_IS_LOYAL_SUBJECT"
						value = 0.5
					}
				}
			}
		}
	}
		
	strategic_region_scores = {
		# Britain is participating in the Caucasian War in a certain capacity early game.
		if = {
			limit = {
				c:GBR ?= this
				exists = c:CIR
				country_rank >= rank_value:great_power
				navy_size >= 50
				scope:region = sr:region_russia
			}
			add = 100
		}

		# Annexation of Hawaii
		if = {
			limit = {
				OR = {
					has_variable = hawaii_interest
					c:USA ?= this
					AND = {
						has_global_variable = gp_hawaii_interest
						country_rank >= rank_value:great_power
						OR = {
							is_in_geographic_region = geographic_region_north_america
							is_in_geographic_region = geographic_region_south_america
						}
					}
				}
				scope:region = sr:region_oceania
			}
			add = 200
		}

		# Sakoku
		if = {
			limit = {
				exists = c:JAP
				c:JAP = { has_law_or_variant = law_type:law_isolationism }
				NOT = { country_rank >= rank_value:great_power }
				scope:region = sr:region_northeast_asia
			}
			add = -50
		}

		# Scramble for Africa
		if = {
			limit = {
				OR = {
					AND = {
						country_rank >= rank_value:great_power
						has_technology_researched = malaria_prevention
					}
					has_journal_entry = je_scramble_for_africa
				}
				scope:region = {
					OR = {
						this = sr:region_nile_basin
						this = sr:region_north_africa
						this = sr:region_west_africa
						this = sr:region_equatorial_africa
						this = sr:region_southern_africa
						this = sr:region_east_africa
					}
				}
			}
			add = 200
		}

		# GBR and RUS care about Afghanistan believe it or not
		if = {
			limit = {
				has_journal_entry = je_great_game_control
				OR = {
					c:GBR ?= this
					exists = c:CIR
					country_rank >= rank_value:great_power
					navy_size >= 50
				}
			}
			add = 500
		}

		if = {
			limit = {
				scope:region = sr:region_oceania
			}
			if = {
				limit = {
					OR = {
						has_variable = hawaii_interest
						AND = {
							exists = c:USA
							c:USA ?= THIS
						}
						AND = {
							has_global_variable = gp_hawaii_interest
							country_rank >= rank_value:great_power
							OR = {
								is_in_geographic_region = geographic_region_north_america
								is_in_geographic_region = geographic_region_south_america
							}
						}
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_nile_basin
			}
			if = {
				limit = { # Certain Great Powers care about Egyptian-Ottoman conflict in the early game
					OR = {
						has_variable = veiled_protectorate_support_ottomans
						has_variable = veiled_protectorate_support_egypt
					}
					OR = {
						navy_size >= 25
						is_adjacent_to_country = c:TUR
						is_adjacent_to_country = c:EGY
					}
				}
				add = 1000
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_north_africa
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_west_africa
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_equatorial_africa
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_southern_africa
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
		if = {
			limit = {
				scope:region = sr:region_east_africa
			}
			if = {
				limit = {
					OR = {
						AND = {
							country_rank >= rank_value:great_power
							has_technology_researched = malaria_prevention
						}
						has_journal_entry = je_scramble_for_africa
					}
				}
				add = 200
			}
		}
	}

	secret_goal_scores = {
		befriend = {
			score = {
				if = { # Try to be on good terms with GBR/USA to avoid intervention
					limit = {
						has_journal_entry = je_reconquista
						scope:target_country = {
							country_rank >= rank_value:great_power
							OR = {
								c:GBR ?= this
								AND = {
									is_in_geographic_region = geographic_region_americas
									is_historical_spanish_american_colony = no
								}
							}
						}
					}
					add = 200
				}
			}
		}
		protect = {
			trigger = {
				OR = {
					country_rank >= rank_value:great_power
					has_strategic_adjacency = scope:target_country
					AND = {
						country_rank >= rank_value:major_power
						scope:target_country = { is_country_type = unrecognized }
					}
				}
			}
		}
		dominate = {
			trigger = {
				OR = {
					country_rank >= rank_value:great_power
					has_strategic_adjacency = scope:target_country
					AND = {
						country_rank >= rank_value:major_power
						scope:target_country = { is_country_type = unrecognized }
					}
				}
			}
		}
	}
}
